# Cell 3: load the ratings and check the basics
import pandas as pd
ratings = pd.read_csv(f'{PROJECT}/raw/AMAZON_FASHION.csv', header=None,
                      names=['asin', 'reviewerID', 'rating', 'timestamp'],
                      dtype={'asin': str, 'reviewerID': str})
print(ratings.head())
print(ratings.dtypes)
print('rows:', len(ratings))
print('missing:'); print(ratings.isna().sum())
print('rating counts:'); print(ratings.rating.value_counts().sort_index())
print('duplicate (user, item) rows:', ratings.duplicated(['reviewerID', 'asin']).sum())
print('date range:', pd.to_datetime(ratings.timestamp, unit='s').agg(['min', 'max']).tolist())

# ===== COLAB CELL 4 (new cell > paste > Shift+Enter) =====
# Cell 4: sparsity. This decides whether collaborative filtering is viable.
uc = ratings.groupby('reviewerID').size()
ic = ratings.groupby('asin').size()
print('users:', len(uc), '| products:', len(ic))
for k in (1, 2, 3, 5, 10):
    print(f'users with >= {k} ratings: {(uc >= k).sum():,}')
for k in (1, 3, 5):
    print(f'products with >= {k} ratings: {(ic >= k).sum():,}')

# ===== COLAB CELL 5 (new cell > paste > Shift+Enter) =====
# Cell 5: load the metadata and see which fields are really filled
import gzip, json
rows = []
with gzip.open(f'{PROJECT}/raw/meta_AMAZON_FASHION.json.gz', 'rt', encoding='utf-8') as f:
    for line in f:
        rows.append(json.loads(line))
meta = pd.DataFrame(rows)
del rows

def filled(v):
    if v is None or (isinstance(v, str) and v == ''):
        return False
    if isinstance(v, float) and v != v:
        return False
    if isinstance(v, (list, dict)) and len(v) == 0:
        return False
    return True

print('shape:', meta.shape)
print('columns:', meta.columns.tolist())
print('duplicate asin:', meta.asin.duplicated().sum())
print((meta.apply(lambda s: s.map(filled).mean()) * 100).round(1).sort_values(ascending=False))

# ===== COLAB CELL 6 (new cell > paste > Shift+Enter) =====
# Cell 6: look at real values, then check how well ratings join to metadata
pd.set_option('display.max_colwidth', 80)
print(meta.sample(3, random_state=0).T)
meta_asins = set(meta.asin)
rated = set(ratings.asin)
print('rated products with metadata:', len(rated & meta_asins), 'of', len(rated))
print('ratings whose product has metadata: %.1f%%' % (ratings.asin.isin(meta_asins).mean() * 100))
